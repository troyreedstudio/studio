FUNCTION 1 — list_open_checks_for_scout (SECURITY DEFINER, reads market_config + checks.coord). Use ST_MakePoint(LNG, LAT):
```sql
create or replace function public.list_open_checks_for_scout(
  p_scout_lat double precision,
  p_scout_lng double precision
) returns setof public.checks
language plpgsql security definer set search_path = public
as $$
declare
  v_radius_m double precision;
  v_scout    geography;
begin
  if p_scout_lat is null or p_scout_lng is null
     or not (p_scout_lat = p_scout_lat) or not (p_scout_lng = p_scout_lng) then
    return;  -- no/NaN location => no jobs (V5)
  end if;
  select dispatch_radius_m into v_radius_m from public.market_config limit 1;  -- Phase 5 single-market; per-market lookup is Phase 7
  v_radius_m := coalesce(v_radius_m, 1500);
  v_scout := ST_SetSRID(ST_MakePoint(p_scout_lng, p_scout_lat), 4326)::geography;
  return query
    select c.*
    from public.checks c
    left join public.venues v on v.id = c.venue_id
    where c.status = 'dispatching' and c.scout_id is null
      and (
        (c.coord is not null and ST_DWithin(c.coord, v_scout, v_radius_m))
        or (v.coord is not null and ST_DWithin(v.coord, v_scout, v_radius_m))
      )
    order by c.created_at asc;
end;
$$;
```

ADD to migration 0012b a service-role timeout sweeper so an unaccepted check ends gracefully (DISP-03), replacing the interim seeker-driven expireUnmatchedCheck.

```sql
create or replace function public.expire_stale_dispatching()
returns int
language plpgsql security definer set search_path = public
as $$
declare v_timeout_s int; v_count int := 0; r record;
begin
  select dispatch_timeout_s into v_timeout_s from public.market_config limit 1;
  v_timeout_s := coalesce(v_timeout_s, 600);
  for r in
    select id from public.checks
    where status = 'dispatching' and scout_id is null
      and updated_at < now() - make_interval(secs => v_timeout_s)
  loop
    -- service role (auth.uid() null) is authorized to drive no_scout (0010 rule)
    perform public.transition_check(r.id, 'no_scout', jsonb_build_object('reason','dispatch_timeout'));
    v_count := v_count + 1;
  end loop;
  return v_count;
end;
$$;
```
Note: the no_scout transition (existing stripe-webhook/cancel flow, Phase 4) releases the Stripe hold so the Seeker is not charged (Pitfall 4). A GPS-rejection re-dispatch resets updated_at, restarting the window.
